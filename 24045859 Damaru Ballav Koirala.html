from scipy.stats import ttest_ind # impotting ttest_ind for t test
df_seperated['Label'].value_counts()
normal = df_seperated[df_seperated['Label'] == 'Normal Traffic']['Flow Duration']
attack = df_seperated[df_seperated['Label'] == 'Attack']['Flow Duration']

t_stat, p_value = ttest_ind(normal, attack) # Perform independent two-sample t-test

print("Null Hypothesis (H0): There is no significant difference in mean Flow Duration between Normal Traffic and Attack.")
print("Alternative Hypothesis (H1): There is a significant difference.\n")

print(f"T-statistic : {t_stat:.4f}")
print(f"P-value     : {p_value:.4f}")

if p_value < 0.05:
    print("\nResult: Reject H0 - There IS a significant difference (p < 0.05)")
else:
    print("\nResult: Fail to reject H0 - No significant difference found (p >= 0.05)")
